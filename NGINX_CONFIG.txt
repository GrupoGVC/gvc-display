# Configuração correta do Nginx para o GVC Display
# Substitua o conteúdo de /etc/nginx/sites-available/gvc-display
# pelo server block abaixo (mantenha o bloco HTTP→HTTPS do Certbot)

server {
    server_name display.drc-gvc.tech 72.61.42.148;
    root /var/www/gvc-display;
    index index.html index.php;

    client_max_body_size 100M;

    # Arquivos estáticos
    location / {
        try_files $uri $uri/ =404;
    }

    # API PHP — passa query string corretamente
    location ~ ^/api/.*\.php$ {
        try_files $uri =404;
        fastcgi_pass unix:/run/php/php8.2-fpm.sock;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param QUERY_STRING    $query_string;
        fastcgi_param HTTP_AUTHORIZATION $http_authorization;
    }

    # tv.php
    location ~ ^/tv(/(.*))?$ {
        fastcgi_pass unix:/run/php/php8.2-fpm.sock;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME /var/www/gvc-display/tv.php;
        fastcgi_param QUERY_STRING    "slug=$2";
        fastcgi_param HTTP_AUTHORIZATION $http_authorization;
    }

    # Outros PHP
    location ~ \.php$ {
        try_files $uri =404;
        fastcgi_pass unix:/run/php/php8.2-fpm.sock;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param QUERY_STRING    $query_string;
        fastcgi_param HTTP_AUTHORIZATION $http_authorization;
    }

    # Segurança
    location ~ /\.(env|git) { deny all; }
    location ~* ^/uploads/.*\.php$ { deny all; }

    # Cache
    location ~* \.html$ { add_header Cache-Control "no-cache, must-revalidate"; expires 0; }
    location ~* \.(js|css)$ { add_header Cache-Control "public, max-age=3600"; expires 1h; }
    location ~* \.(png|jpg|jpeg|gif|ico|svg|mp4|woff|woff2|ttf)$ { add_header Cache-Control "public, max-age=604800"; expires 7d; }

    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/display.drc-gvc.tech/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/display.drc-gvc.tech/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}
